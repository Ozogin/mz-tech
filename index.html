var score = 0; var gameInterval = null; var gameActive = false; var audioCtx = null; var ambientOsc = null; // Сверхточный таймер аптайма реального времени прямо на экране! var secondsPassed = 0; function runRealtimeClock() { secondsPassed++; var h = Math.floor(secondsPassed / 3600); var m = Math.floor((secondsPassed % 3600) / 60); var s = secondsPassed % 60; document.getElementById("b-uptime").innerText = h + "ч " + m + "мин " + s + "сек"; } setInterval(runRealtimeClock, 1000); function initAtmosphere() { if (audioCtx) return; audioCtx = new (window.AudioContext || window.webkitAudioContext)(); ambientOsc = audioCtx.createOscillator(); var ambientGain = audioCtx.createGain(); ambientOsc.type = 'triangle'; ambientOsc.frequency.setValueAtTime(65.41, audioCtx.currentTime); ambientGain.gain.setValueAtTime(0.06, audioCtx.currentTime); ambientOsc.connect(ambientGain); ambientGain.connect(audioCtx.destination); ambientOsc.start(); document.getElementById('audio-btn').innerText = "🔮 Атмосферная фоновая музыка успешно включена!"; } function playCatchSound() { if(!audioCtx) return; var osc = audioCtx.createOscillator(); var gainNode = audioCtx.createGain(); osc.type = 'sine'; osc.frequency.setValueAtTime(220, audioCtx.currentTime); osc.frequency.exponentialRampToValueAtTime(60, audioCtx.currentTime + 0.2); gainNode.gain.setValueAtTime(0.06, audioCtx.currentTime); gainNode.gain.linearRampToValueAtTime(0.001, audioCtx.currentTime + 0.2); osc.connect(gainNode); gainNode.connect(audioCtx.destination); osc.start(); osc.stop(audioCtx.currentTime + 0.2); } function movePumpkin() { if(!gameActive) return; var p = document.getElementById('pumpkin'); var x = Math.floor(Math.random() * 320); var y = Math.floor(Math.random() * 140); p.style.left = x + 'px'; p.style.top = y + 'px'; } document.getElementById('audio-btn').addEventListener('click', initAtmosphere); document.getElementById('btn-start').addEventListener('click', function() { score = 0; document.getElementById('score').innerText = score; gameActive = true; if(gameInterval) clearInterval(gameInterval); movePumpkin(); gameInterval = setInterval(movePumpkin, 750); }); document.getElementById('btn-stop').addEventListener('click', function() { gameActive = false; if(gameInterval) clearInterval(gameInterval); var p = document.getElementById('pumpkin'); p.style.left = '45%'; p.style.top = '40%'; }); document.getElementById('pumpkin').addEventListener('click', function() { if(!gameActive) return; score++; document.getElementById('score').innerText = score; try { playCatchSound(); } catch(e){} movePumpkin(); });
